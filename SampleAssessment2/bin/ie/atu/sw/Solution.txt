class AbsoluteOperation implements DoubleOperation {
    private final double threshold;
    
    public AbsoluteOperation(double threshold) {
        this.threshold = threshold;
    }
    
    @Override
    public double process(double value) {
        System.out.println("AbsoluteOperation: processing " + value);
        return value >= threshold ? Math.abs(value) : value;
    }
}

/**
 * Factory creates operations based on Activity level
 * Hides construction complexity
 */
class DoubleOperationFactory {
    private static final double DEFAULT_THRESHOLD = 42.0d;
    
    public static DoubleOperation createAbsolute() {
        System.out.println("Factory: Creating AbsoluteOperation");
        return new AbsoluteOperation(DEFAULT_THRESHOLD);
    }
    
    public static DoubleOperation createAbsolute(double threshold) {
        System.out.println("Factory: Creating AbsoluteOperation with threshold " + threshold);
        return new AbsoluteOperation(threshold);
    }
    
    public static DoubleOperation createAbsoluteLog(Activity activity) {
        System.out.println("Factory: Creating AbsoluteLogOperation for " + activity);
        return new AbsoluteLogOperation(DEFAULT_THRESHOLD, activity.level());
    }
    
    public static DoubleOperation createAbsoluteReverse() {
        System.out.println("Factory: Creating AbsoluteReverseOperation");
        return new AbsoluteReverseOperation(DEFAULT_THRESHOLD);
    }
    
    public static DoubleOperation createAbsoluteLogBitFlip(Activity activity) {
        System.out.println("Factory: Creating AbsoluteLogBitFlipOperation for " + activity);
        return new AbsoluteLogBitFlipOperation(DEFAULT_THRESHOLD, activity.level());
    }
    
    // Smart factory method that chooses based on activity
    public static DoubleOperation createForActivity(Activity activity) {
        System.out.println("Factory: Creating operation for activity " + activity);
        
        return switch (activity) {
            case NONE, LOW -> createAbsolute();
            case MEDIUM, ELEVATED -> createAbsoluteLog(activity);
            case HIGH, VERY_HIGH -> createAbsoluteLogBitFlip(activity);
        };
    }
}

class DoubleOps {
    private static DoubleOps instance;
    private Map<Activity, DoubleOperation> operationRegistry;
    
    // Private constructor for Singleton
    private DoubleOps() {
        this.operationRegistry = new HashMap<>();
        System.out.println("DoubleOps: Singleton instance created");
    }
    
    public static synchronized DoubleOps getInstance() {
        if (instance == null) {
            instance = new DoubleOps();
        }
        return instance;
    }
    
    /**
     * Register an operation for an activity level
     * This is DEPENDENCY INJECTION
     */
    public void registerOperation(Activity activity, DoubleOperation operation) {
        System.out.println("DoubleOps: Registering operation for " + activity);
        operationRegistry.put(activity, operation);
    }
    
    /**
     * Process a value using the operation registered for this activity
     * REPLACES all the getAbsoluteXXX methods with ONE method!
     */
    public double process(Activity activity, double value) {
        System.out.println("DoubleOps: Processing value " + value + " for " + activity);
        
        DoubleOperation operation = operationRegistry.get(activity);
        
        if (operation == null) {
            System.out.println("No operation registered for " + activity + ", using default");
            return value;
        }
        
        return operation.process(value);
    }
    
    // Convenience method for simple absolute value
    public double processAbsolute(double value) {
        return DoubleOperationFactory.createAbsolute().process(value);
    }
}
